class AuthSystem {
    + keyStore : SecureKeyStore
    + publicKey : uint64
    + privateKey : uint64
    + usedEmergencyKeyIDs : Map
    + verificationLog : cell
    + keyChain : cell
    + currentKeyIndex : int
    + disclosedUntilIndex : int
    + messageBuffer : cell
    + DELAY_PACKETS : int
    + CHAIN_LENGTH : int
    --
    + AuthSystem(delayPackets, chainLength)
    + issueEmergencyKey(keyID, icao) : AuthKey
    + getNextNormalKey() : AuthKey
    + discloseNormalKey(keyIndex)
    + computeAuthMAC(key, message) : mac, ms
    + verifyAuth(key, message, mac) : isValid, ms, reason
    + verifyProximity(distance, delay) : granted, dist
    --
    - generateKeyChain(len) : cell
    - verifyKeyChainIntegrity(keyIndex) : bool
    - processBufferForIndex(keyIndex)
    - verifyInstant(key, message, mac) : isValid, reason
    - verifyDelayed(key, message, mac) : isValid, ms, reason
}

class AuthKey {
    + keyID : int32
    + keyMaterial : uint8[32]
    + issueTime : int64
    + signature : uint8[16]
    + used : bool
    --
    + AuthKey(keyID, keyMaterial, issueTime, signature)
}

class SecureKeyStore {
    - keyStore : Map
    - usedKeyIDs : Map
    + hardwareProtected : bool
    --
    + SecureKeyStore()
    + storeKey(key)
    + useKey(keyID) : AuthKey
    + peekKey(keyID) : AuthKey
    + isProtected() : bool
}

class ProximityVerifier {
    + {static} LIGHT_SPEED : double
    + {static} MAX_RANGE_M : double
    + {static} BIT_COUNT : int
    + {static} BASE_PROC_TIME : double
    --
    + {static} buildRegisters() : reg0, reg1
    + {static} measureRTT(reg0, reg1, dist, delay) : time, bits
    + {static} computeRange(totalTime) : double
    + {static} checkAccess(reg0, reg1, dist, delay) : granted, ...
}

' ===================================================
' RELATIONSHIPS
' ===================================================

' Composition: AuthSystem owns its store
AuthSystem *--> SecureKeyStore : owns >

' AuthSystem creates and manages AuthKey objects
AuthSystem ..> AuthKey : creates & stores >

' SecureKeyStore holds AuthKey records
SecureKeyStore o--> AuthKey : holds >

' AuthSystem delegates all crypto to utilities
AuthSystem ..> computeMAC    : uses >
AuthSystem ..> signVerify     : uses >
AuthSystem ..> generateKeypair : uses at startup >

' AuthSystem calls ProximityVerifier for RTT checks
AuthSystem ..> ProximityVerifier : delegates proximity check >

' Entry point drives the system
AdsbAuthMain ..> AuthSystem         : creates & drives >
AdsbAuthMain ..> runProximityTest   : calls >
AdsbAuthMain ..> runProximityMonteCarlo : calls >

' Proximity scripts use the static verifier
runProximityTest       ..> ProximityVerifier : calls static methods >
runProximityMonteCarlo ..> ProximityVerifier : calls static methods >

' Result codes returned by AuthSystem
AuthSystem ..> VerificationResult : returns >
